type Container struct {
        srv    *http.Server
        cfg    *config.Config
        db     *postgres.DB
        tracer *telemetry.OTelProviders
}

func New(cfg *config.Config) (*Container, error) {
        connectCtx, cancel := context.WithTimeout(context.Background(), cfg.DatabaseConnectTimeout)
        defer cancel()

        provider, err := telemetry.NewOTel(connectCtx, cfg)
        if err != nil {
                return nil, fmt.Errorf("failed to init tracer: %w", err)
        }
        logger := telemetry.NewLogger(config.ParseLodLevel(cfg.LogLevel))
        meter := provider.Meter.Meter(cfg.OtelServiceName)

        db, err := postgres.NewDB(connectCtx, cfg)
        if err != nil {
                return nil, fmt.Errorf("failed to load storage: %v", err)
        }

        tripRepo := trip.NewTripRepository(db.Pool, cfg.DatabaseQueryTimeout)
        tm := txManager.NewTransactionManager(db.Pool)

        createUc := createUsecase.NewCreateUseCase(tripRepo, tm, logger, meter)
        finishUc := finishUsecase.NewFinishUseCase(tripRepo, tm, logger, meter)
        getUc := getUsecase.NewGetUseCase(tripRepo, logger)

        createH := create.NewHandler(createUc)
        finishH := finish.NewHandler(finishUc)
        getH := get.NewHandler(getUc)

        healthH := health.NewHandler(db.Pool)

        si := server.NewServerInterface(getH, createH, finishH, healthH)
        r := chi.NewRouter()

        r.Use(middleware.Recover(logger))
        r.Use(middleware.Tracing())
        r.Use(middleware.Logging(logger))
        r.Use(middleware.Metrics(meter))

        api.HandlerFromMux(si, r)
        srv := &http.Server{
                Addr:              cfg.HttpAddr,
                Handler:           r,
                ReadTimeout:       cfg.HttpReadTimeout,
                ReadHeaderTimeout: cfg.HttpReadHeaderTimeout,
                WriteTimeout:      cfg.HttpWriteTimeout,
                IdleTimeout:       cfg.HttpIdleTimeout,
        }
        return &Container{srv: srv, cfg: cfg, db: db, tracer: provider}, nil
}

func (c *Container) Run() {
        defer c.db.Close()
        go func() {
                log.Printf("Starting server on %s", c.srv.Addr)
                if err := c.srv.ListenAndServe(); err != nil {
                        log.Printf("Server error: %v", err)
                }
        }()

        quit := make(chan os.Signal, 1)
        signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
        <-quit

        log.Println("shutting down gracefully")

        shotdownCtx, cancel := context.WithTimeout(context.Background(), c.cfg.ShutdownTimeout)
        defer cancel()

        if c.tracer != nil {
                defer c.tracer.Shutdown(shotdownCtx)
        }

        if err := c.srv.Shutdown(shotdownCtx); err != nil {
                log.Printf("Server forced to shutdown: %v", err)
        }
        log.Println("Server exited gracefully")
}

package domainerror

import (
        "encoding/json"
        "errors"
        "net/http"

        "github.com/adeladnagulov/TripGo/api"
        "github.com/adeladnagulov/TripGo/internal/domain/trip"
)

func Problem(err error, r *http.Request) (int, api.Problem) {
        if err == nil {
                return http.StatusOK, api.Problem{}
        }
        host := r.URL.Host
        typeConflict := "about:blank"
        switch {
        case isValidationError(err):
                detail := "invalid request body or parameters"
                return http.StatusBadRequest, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusBadRequest),
                        Status:   http.StatusBadRequest,
                        Detail:   &detail,
                        Code:     "invalid_request",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrDriverBusy):
                detail := "driver already has active trip"
                return http.StatusConflict, api.CreateTripConflict{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusConflict),
                        Status:   http.StatusConflict,
                        Detail:   &detail,
                        Code:     "driver_busy",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrTripCompleted):
                detail := "trip is already over"
                return http.StatusConflict, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusConflict),
                        Status:   http.StatusConflict,
                        Detail:   &detail,
                        Code:     "trip_completed",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrNotFound):
                detail := "trip with an id does not exist"
                return http.StatusNotFound, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusNotFound),
                        Status:   http.StatusNotFound,
                        Detail:   &detail,
                        Code:     "trip_not_found",
                        Instance: &host,
                }
        }

        detail := "internal server error"
        return http.StatusInternalServerError, api.InternalError{
                Type:     typeConflict,
                Title:    http.StatusText(http.StatusInternalServerError),
                Status:   http.StatusInternalServerError,
                Detail:   &detail,
                Code:     "internal_error",
                Instance: &host,
        }
}

func isValidationError(err error) bool {
        var syntaxErr *json.SyntaxError
        var unmarshalErr *json.UnmarshalTypeError
        if errors.As(err, &syntaxErr) || errors.As(err, &unmarshalErr) {
                return true
        }
        return false
}

func Logging(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        lrw := &statusResponseWriter{ResponseWriter: w, status: http.StatusOK}
                        next.ServeHTTP(lrw, r)

                        duration := time.Since(start)
                        level := slog.LevelInfo
                        if lrw.status >= 500 {
                                level = slog.LevelError
                        }

                        routePattern := ""
                        if routeCtx := chi.RouteContext(r.Context()); routeCtx != nil {
                                routePattern = routeCtx.RoutePattern()
                        }

                        logger.LogAttrs(r.Context(), level, "request handled",
                                slog.String("service", "trip-service"),
                                slog.String("method", r.Method),
                                slog.String("route", routePattern),
                                slog.Int("status", lrw.status),
                                slog.Int64("duration_ms", duration.Milliseconds()),
                        )
                })
        }
}

type statusResponseWriter struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (lrw *statusResponseWriter) WriteHeader(code int) {
        if !lrw.wroteHeader {
                lrw.status = code
                lrw.wroteHeader = true
                lrw.ResponseWriter.WriteHeader(code)
        }
}

func (lrw *statusResponseWriter) Write(b []byte) (int, error) {
        if !lrw.wroteHeader {
                lrw.WriteHeader(http.StatusOK)
        }
        return lrw.ResponseWriter.Write(b)
}

import (
        "net/http"
        "time"

        "github.com/go-chi/chi/v5"
        "go.opentelemetry.io/otel/attribute"
        "go.opentelemetry.io/otel/metric"
)

func Metrics(meter metric.Meter) func(http.Handler) http.Handler {
        durationHist, _ := meter.Float64Histogram(
                "http.server.request.duration",
                metric.WithUnit("s"),
                metric.WithDescription("how long did the answer take"),
        )

        activeReq, _ := meter.Int64UpDownCounter(
                "http.server.active_requests",
                metric.WithUnit("{request}"),
                metric.WithDescription("how many requests are being processed right now"),
        )

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        routePattern := chi.RouteContext(r.Context()).RoutePattern()
                        activeReq.Add(r.Context(), 1, metric.WithAttributes(
                                attribute.String("http.request.method", r.Method),
                                attribute.String("http.route", routePattern),
                        ))
                        start := time.Now()
                        lrw := &statusResponseWriter{ResponseWriter: w, status: http.StatusOK}
                        next.ServeHTTP(lrw, r)

                        duration := time.Since(start).Seconds()
                        durationHist.Record(r.Context(), duration, metric.WithAttributes(
                                attribute.String("http.request.method", r.Method),
                                attribute.String("http.route", routePattern),
                                attribute.Int("http.response.status_code", lrw.status),
                        ))

                        activeReq.Add(r.Context(), -1, metric.WithAttributes(
                                attribute.String("http.request.method", r.Method),
                                attribute.String("http.route", routePattern),
                        ))
                })
        }
}

func Recover(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                if err := recover(); err != nil {
                                        logger.ErrorContext(r.Context(), "panic recovered",
                                                "error", err,
                                                "stack", string(debug.Stack()),
                                        )

                                        status, problem := domainerror.Problem(errors.New("internal server error"), r)
                                        response.Problem(w, status, problem)
                                }
                        }()
                        next.ServeHTTP(w, r)
                })
        }
}

import (

        "github.com/adeladnagulov/TripGo/internal/config"
)

type DB struct {
        Pool *pgxpool.Pool
}

func NewDB(ctx context.Context, cfg *config.Config) (*DB, error) {
        pgxCfg, err := pgxpool.ParseConfig(cfg.DatabaseUrl)
        if err != nil {
                return nil, fmt.Errorf("parse pgx config: %w", err)
        }
        pgxCfg.MaxConns = int32(cfg.DatabaseMaxConns) //nolint:gosec
        pgxCfg.MinConns = int32(cfg.DatabaseMinConns) //nolint:gosec
        pgxCfg.MaxConnLifetime = cfg.DatabaseMaxConnsLifetime
        pgxCfg.ConnConfig.ConnectTimeout = cfg.DatabaseConnectTimeout
        pgxCfg.ConnConfig.Tracer = otelpgx.NewTracer()

        pool, err := pgxpool.NewWithConfig(ctx, pgxCfg)
        if err != nil {
                return nil, fmt.Errorf("create pool: %w", err)
        }

        pingCtx, pingCancel := context.WithTimeout(context.Background(), cfg.DatabaseConnectTimeout)
        defer pingCancel()
        if err = pool.Ping(pingCtx); err != nil {
                pool.Close()
                return nil, fmt.Errorf("connection from pool: %w", err)
        }

        return &DB{Pool: pool}, nil
}

func (db *DB) Close() {
        if db.Pool != nil {
                db.Pool.Close()
        }
}

func (db *DB) PingTrip(ctx context.Context) error {
        return db.Pool.Ping(ctx)
}

package txManager

import (
        "context"
        "fmt"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"
)

type contextKey string

const TxKey contextKey = "transaction"

type TransactionManager struct {
        pool *pgxpool.Pool
}

func NewTransactionManager(pool *pgxpool.Pool) *TransactionManager {
        return &TransactionManager{
                pool: pool,
        }
}

func (tm *TransactionManager) Do(ctx context.Context, fn func(ctx context.Context) error) (err error) {
        if _, ok := ctx.Value(TxKey).(pgx.Tx); ok {
                return fn(ctx)
        }

        tx, err := tm.pool.Begin(ctx)
        if err != nil {
                return fmt.Errorf("starts a transaction: %w", err)
        }

        defer func() {
                if p := recover(); p != nil {
                        _ = tx.Rollback(ctx)
                        panic(p)
                } else if err != nil {
                        _ = tx.Rollback(ctx)
                } else {
                        _ = tx.Commit(ctx)
                }
        }()

        ctx = context.WithValue(ctx, TxKey, tx)
        return fn(ctx)
}

        "log/slog"
        "os"
        "time"

        "go.opentelemetry.io/otel/trace"
)

func NewLogger(level slog.Level) *slog.Logger {
        opts := &slog.HandlerOptions{
                Level: level,
                ReplaceAttr: func(groups []string, a slog.Attr) slog.Attr {
                        if a.Key == slog.TimeKey {
                                if t, ok := a.Value.Any().(time.Time); ok {
                                        a.Value = slog.StringValue(t.UTC().Format("2006-01-02T15:04:05.000Z"))
                                }
                        }
                        return a
                },
        }

        handler := slog.NewJSONHandler(os.Stdout, opts)
        return slog.New(&traceHandler{inner: handler})
}

type traceHandler struct {
        inner slog.Handler
}

func (h *traceHandler) Enabled(ctx context.Context, level slog.Level) bool {
        return h.inner.Enabled(ctx, level)
}

func (h *traceHandler) Handle(ctx context.Context, r slog.Record) error {
        if span := trace.SpanFromContext(ctx); span.SpanContext().IsValid() {
                sc := span.SpanContext()
                r.AddAttrs(
                        slog.String("trace_id", sc.TraceID().String()),
                        slog.String("span_id", sc.SpanID().String()),
                )
        }
        return h.inner.Handle(ctx, r)
}

func (h *traceHandler) WithAttrs(attrs []slog.Attr) slog.Handler {
        return &traceHandler{inner: h.inner.WithAttrs(attrs)}
}

func (h *traceHandler) WithGroup(name string) slog.Handler {
        return &traceHandler{inner: h.inner.WithGroup(name)}
}

        semconv "go.opentelemetry.io/otel/semconv/v1.37.0"

        "github.com/adeladnagulov/TripGo/internal/config"
)

type OTelProviders struct {
        Tracer *trace.TracerProvider
        Meter  *metric.MeterProvider
}

func NewOTel(ctx context.Context, cfg *config.Config) (*OTelProviders, error) {
        p := &OTelProviders{}

        res, err := resource.New(ctx,
                resource.WithFromEnv(),
                resource.WithAttributes(
                        semconv.ServiceNameKey.String(cfg.OtelServiceName),
                ),
        )
        if err != nil {
                return nil, err
        }

        endpointHost := cfg.OtelExporterEndpoint
        if u, err := url.Parse(cfg.OtelExporterEndpoint); err == nil && u.Host != "" {
                endpointHost = u.Host
        }

        traceExp, err := otlptracehttp.New(ctx,
                otlptracehttp.WithEndpoint(endpointHost),
                otlptracehttp.WithInsecure(),
        )
        if err != nil {
                slog.Warn("trace exporter init failed, traces disabled", "error", err)
        } else {
                p.Tracer = trace.NewTracerProvider(
                        trace.WithResource(res),
                        trace.WithBatcher(traceExp),
                        trace.WithSampler(trace.ParentBased(trace.AlwaysSample())),
                )
                otel.SetTracerProvider(p.Tracer)
        }

        metricExp, err := otlpmetrichttp.New(ctx,
                otlpmetrichttp.WithEndpoint(endpointHost),
                otlpmetrichttp.WithInsecure(),
        )
        if err != nil {
                slog.Warn("metric exporter init failed", "error", err)
        } else {
                p.Meter = metric.NewMeterProvider(
                        metric.WithResource(res),
                        metric.WithReader(metric.NewPeriodicReader(metricExp)),
                )
                otel.SetMeterProvider(p.Meter)
        }

        otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
                propagation.TraceContext{},
                propagation.Baggage{},
        ))

        return p, nil
}

func (tp *OTelProviders) Shutdown(ctx context.Context) {
        if tp.Tracer != nil {
                _ = tp.Tracer.Shutdown(ctx)
        }
        if tp.Meter != nil {
                _ = tp.Meter.Shutdown(ctx)
        }
}

package trip

import (
        "context"
        "errors"
        "fmt"
        "time"

        sq "github.com/Masterminds/squirrel"
        "github.com/google/uuid"
        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgconn"
        "github.com/jackc/pgx/v5/pgxpool"

        "github.com/adeladnagulov/TripGo/internal/domain/trip"
        txManager "github.com/adeladnagulov/TripGo/internal/infra/postgres/transaction"
)

type Repository struct {
        pool         *pgxpool.Pool
        queryTimeout time.Duration
}

func NewTripRepository(pool *pgxpool.Pool, queryTimeout time.Duration) *Repository {
        return &Repository{
                pool:         pool,
                queryTimeout: queryTimeout,
        }
}

type dbExecutor interface {
        Exec(ctx context.Context, sql string, arguments ...any) (commandTag pgconn.CommandTag, err error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
}

func getDbExecutor(ctx context.Context, pool *pgxpool.Pool) dbExecutor {
        if tx, ok := ctx.Value(txManager.TxKey).(pgx.Tx); ok {
                return tx
        }
        return pool
}

func (r *Repository) Ping(ctx context.Context) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        return r.pool.Ping(queryCtx)
}

func (r *Repository) Create(ctx context.Context, t *trip.Trip) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)

        tripBuilder := sq.Insert("trips").
                Columns(
                        "id", "user_id", "driver_id",
                        "start_latitude", "start_longitude", "end_latitude", "end_longitude",
                        "price", "status",
                        "started_at", "created_at", "updated_at",
                ).
                Values(
                        t.Id, t.UserId, t.DriverId,
                        t.StartPoint.Lat, t.StartPoint.Lon, t.EndPoint.Lat, t.EndPoint.Lon,
                        t.Price, t.Status,
                        t.StartedAt, time.Now(), time.Now(),
                ).
                PlaceholderFormat(sq.Dollar)

        query, args, err := tripBuilder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip query: %w", err)
        }
        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                var pgErr *pgconn.PgError
                if errors.As(err, &pgErr) && pgErr.Code == "23505" {
                        return trip.ErrDriverBusy
                }
                return fmt.Errorf("failed exec trips: %w", err)
        }

        historyBulder := sq.Insert("trip_status_history").
                Columns("trip_id", "from_status", "to_status", "reason", "changed_at").
                Values(t.Id, nil, t.Status, "trip created", t.StartedAt).
                PlaceholderFormat(sq.Dollar)

        query, args, err = historyBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip_status_history query: %w", err)
        }
        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                return fmt.Errorf("failed exec trip_status_history: %w", err)
        }

        return nil
}

func (r *Repository) Save(ctx context.Context, t *trip.Trip) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)

        oldBulder := sq.Select("status").
                From("trips").
                Where(sq.Eq{"id": t.Id}).
                PlaceholderFormat(sq.Dollar)
        query, args, err := oldBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde old_trip query: %w", err)
        }
        var oldStatus string
        err = dbExc.QueryRow(queryCtx, query, args...).Scan(&oldStatus)
        if err != nil {
                if err == pgx.ErrNoRows {
                        return trip.ErrNotFound
                }
                return fmt.Errorf("failed QueryRow old status: %w", err)
        }

        tripBulder := sq.Update("trips").
                Set("status", t.Status).
                Set("finished_at", t.FinishedAt).
                Set("updated_at", t.FinishedAt).
                Where(sq.Eq{"id": t.Id}).
                PlaceholderFormat(sq.Dollar)
        query, args, err = tripBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip query: %w", err)
        }
        tag, err := dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                return fmt.Errorf("failed exec trips: %w", err)
        }
        if tag.RowsAffected() == 0 {
                return trip.ErrNotFound
        }

        historyBulder := sq.Insert("trip_status_history").
                Columns("trip_id", "from_status", "to_status", "reason", "changed_at").
                Values(t.Id, oldStatus, t.Status, "trip finished", time.Now()). // В будующем, помменять "trip finished" если не только за конца поезди будет
                PlaceholderFormat(sq.Dollar)

        query, args, err = historyBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip_status_history query: %w", err)
        }
        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                return fmt.Errorf("failed exec trip_status_history: %w", err)
        }

        return nil
}

func (r *Repository) GetByID(ctx context.Context, id uuid.UUID) (*trip.Trip, error) {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)

        tripBulder := sq.Select(
                "id", "user_id", "driver_id",
                "start_latitude", "start_longitude", "end_latitude", "end_longitude",
                "price", "status",
                "started_at", "finished_at",
        ).
                From("trips").
                Where(sq.Eq{"id": id}).
                PlaceholderFormat(sq.Dollar)

        query, args, err := tripBulder.ToSql()
        if err != nil {
                return nil, fmt.Errorf("builde trip query: %w", err)
        }

        t := trip.Trip{}
        err = dbExc.QueryRow(queryCtx, query, args...).Scan(
                &t.Id,
                &t.UserId,
                &t.DriverId,
                &t.StartPoint.Lat,
                &t.StartPoint.Lon,
                &t.EndPoint.Lat,
                &t.EndPoint.Lon,
                &t.Price,
                &t.Status,
                &t.StartedAt,
                &t.FinishedAt,
        )
        if err != nil {
                if err == pgx.ErrNoRows {
                        return nil, trip.ErrNotFound
                }
                return nil, fmt.Errorf("failed QueryRow trip: %w", err)
        }

        return &t, nil
}

package create

import (
        "context"
        "fmt"
        "log/slog"
        "time"

        "github.com/google/uuid"
        "go.opentelemetry.io/otel/metric"

        "github.com/adeladnagulov/TripGo/internal/domain/trip"
)

type CreateUseCase struct {
        repo         tripCreator
        tm           txManager
        logger       *slog.Logger
        tripsCreated metric.Int64Counter
}

func NewCreateUseCase(repo tripCreator, tm txManager, logger *slog.Logger, meter metric.Meter) *CreateUseCase {
        tripsCreated, err := meter.Int64Counter(
                "tripgo.trips.created",
                metric.WithDescription("created trips"),
        )
        if err != nil {
                logger.Error("failed to create trips_created metric", "error", err)
        }
        return &CreateUseCase{
                repo:         repo,
                tm:           tm,
                logger:       logger,
                tripsCreated: tripsCreated,
        }
}

func (s *CreateUseCase) CreateTrip(ctx context.Context, data trip.TripData) (*trip.Trip, error) {
        t := trip.Trip{
                Id:         uuid.New(),
                UserId:     data.UserId,
                DriverId:   data.DriverId,
                StartPoint: data.StartPoint,
                EndPoint:   data.EndPoint,
                Price:      data.Price,
                Status:     trip.Active,
                StartedAt:  time.Now(),
        }

        err := s.tm.Do(ctx, func(ctx context.Context) error {
                return s.repo.Create(ctx, &t)
        })
        if err != nil {
                s.logger.WarnContext(ctx, "create trip error",
                        "id", t.Id,
                        "driver_id", t.DriverId,
                )
                return nil, fmt.Errorf("create trip error: %w", err)
        }
        s.tripsCreated.Add(ctx, 1)
        s.logger.InfoContext(ctx, "trip created successfully",
                "id", t.Id,
                "driver_id", t.DriverId,
                "user_id", t.UserId,
                "price", t.Price,
        )
        return &t, nil
}

package finish

import (
        "context"
        "fmt"
        "log/slog"
        "time"

        "github.com/google/uuid"
        "go.opentelemetry.io/otel/metric"

        "github.com/adeladnagulov/TripGo/internal/domain/trip"
)

type FinishUseCase struct {
        repo           tripFinisher
        tm             txManager
        logger         *slog.Logger
        tripsComplited metric.Int64Counter
}

func NewFinishUseCase(repo tripFinisher, tm txManager, logger *slog.Logger, meter metric.Meter) *FinishUseCase {
        tripsComplited, err := meter.Int64Counter(
                "tripgo.trips.complete",
                metric.WithDescription("completed trips"),
        )
        if err != nil {
                logger.Error("failed to create trips_complited metric", "error", err)
        }
        return &FinishUseCase{
                repo:           repo,
                tm:             tm,
                logger:         logger,
                tripsComplited: tripsComplited,
        }
}

func (s *FinishUseCase) FinishTrip(ctx context.Context, id uuid.UUID) (*trip.Trip, error) {
        finishTrip, err := s.repo.GetByID(ctx, id)
        if err != nil {
                return nil, fmt.Errorf("get finish trip id=%v error: %w", id, err)
        }

        err = finishTrip.Finish(time.Now())
        if err != nil {
                s.logger.WarnContext(ctx, "trip already complited",
                        "id", finishTrip.Id,
                )
                return nil, err
        }

        err = s.tm.Do(ctx, func(ctx context.Context) error {
                err := s.repo.Save(ctx, finishTrip)
                if err != nil {
                        return fmt.Errorf("save finish trip id=%v error: %w", id, err)
                }
                return nil
        })
        if err != nil {
                return nil, err
        }
        s.tripsComplited.Add(ctx, 1)
        s.logger.WarnContext(ctx, "trip finished successfully",
                "id", finishTrip.Id,
                "status", finishTrip.Status,
        )
        return finishTrip, nil
}
