type Container struct {
        srv    *http.Server
        cfg    *config.Config
        db     *postgres.DB
        tracer *telemetry.OTelProviders
        wp     *workerpool.WorkerPool
        scan   *scanner.Scanner
}

func New(cfg *config.Config) (*Container, error) {
        connectCtx, cancel := context.WithTimeout(context.Background(), cfg.DatabaseConnectTimeout)
        defer cancel()

        provider, err := telemetry.NewOTel(connectCtx, cfg)
        if err != nil {
                return nil, fmt.Errorf("failed to init tracer: %w", err)
        }
        logger := telemetry.NewLogger(config.ParseLodLevel(cfg.LogLevel))
        meter := provider.Meter.Meter(cfg.OtelServiceName)
        scanMeter, err := metrics.NewScannerMetrics(meter)
        if err != nil {
                return nil, fmt.Errorf("failed create scanMeter: %v", err)
        }
        var wp *workerpool.WorkerPool
        poolMetrics, err := metrics.NewPoolMetrics(
                meter,
                func() int { return wp.GetQueueSize() },
                func() int64 { return wp.GetActiveWorkers() },
        )
        if err != nil {
                return nil, fmt.Errorf("failed create poolMetrics: %v", err)
        }

        db, err := postgres.NewDB(connectCtx, cfg)
        if err != nil {
                return nil, fmt.Errorf("failed to load storage: %v", err)
        }

        tripRepo := trip.NewTripRepository(db.Pool, cfg.DatabaseQueryTimeout)
        tm := txManager.NewTransactionManager(db.Pool, cfg.DatabaseQueryTimeout)

        inQueue := currentlyinqueue.NewInQueue()
        gateWay := push.NewGateway(cfg.PushHTTPURL, cfg.PushRequestTimeout)
        wp = workerpool.NewWorkerPool(cfg.PositionQueueSize, inQueue, gateWay, cfg.PushRequestTimeout, cfg.PositionWorkers, logger, poolMetrics)
        tick := ticker.NewRealTicker(cfg.PositionCheckInterval)
        scan := scanner.NewScanner(cfg.PositionCheckInterval, cfg.PositionStaleAfter, tripRepo, wp, inQueue, tick, logger, scanMeter)

        createUc := createUsecase.NewCreateUseCase(tripRepo, tm, logger, meter)
        finishUc := finishUsecase.NewFinishUseCase(tripRepo, tm, logger, meter)
        getUc := getUsecase.NewGetUseCase(tripRepo, logger)
        savePosUc := savePositionUsecase.NewSavePositionUseCase(tripRepo, tm, logger, inQueue, meter)
        getPossUc := getPositionsUsecase.NewGetUseCase(tripRepo, logger)

        createH := create.NewHandler(createUc)
        finishH := finish.NewHandler(finishUc)
        getH := get.NewHandler(getUc)
        healthH := health.NewHandler(db.Pool)
        createPosH := createposition.NewHandler(savePosUc)
        getPossH := getpositions.NewHandler(getPossUc)
        si := server.NewServerInterface(getH, createH, finishH, healthH, createPosH, getPossH)

        specPath := "contracts/openapi/trip-service.openapi.yaml"
        specData, err := os.ReadFile(specPath)
        if err != nil {
                log.Printf("Failed to read openapi spec from %s: %v", specPath, err)
        }
        loader := openapi3.NewLoader()
        swagger, err := loader.LoadFromData(specData)
        if err != nil {
                log.Printf("Failed to load swagger spec: %v", err)
        }
        if err := swagger.Validate(loader.Context); err != nil {
                log.Printf("Swagger spec is invalid: %v", err)
        }
        swagger.Servers = nil

        r := chi.NewRouter()
        openAPIRequestValidator, err := middleware.OpenAPIRequestValidator(swagger)
        if err != nil {
                return nil, fmt.Errorf("failed to init OpenAPI validator: %w", err)
        }
        r.Use(openAPIRequestValidator)
        r.Use(middleware.Recover(logger))
        r.Use(middleware.Logging(logger))
        api.HandlerWithOptions(si, api.ChiServerOptions{
                BaseRouter: r,
                Middlewares: []api.MiddlewareFunc{
                        api.MiddlewareFunc(middleware.Tracing()),
                        api.MiddlewareFunc(middleware.Metrics(meter)),
                },
                ErrorHandlerFunc: response.HandleAPIError,
        })

        srv := &http.Server{
                Addr:              cfg.HttpAddr,
                Handler:           r,
                ReadTimeout:       cfg.HttpReadTimeout,
                ReadHeaderTimeout: cfg.HttpReadHeaderTimeout,
                WriteTimeout:      cfg.HttpWriteTimeout,
                IdleTimeout:       cfg.HttpIdleTimeout,
        }
        return &Container{srv: srv, cfg: cfg, db: db, tracer: provider, wp: wp, scan: scan}, nil
}

func (c *Container) Run() {
        srvErrCh := make(chan error, 1)
        go func() {
                log.Printf("Starting server on %s", c.srv.Addr)
                if err := c.srv.ListenAndServe(); err != nil {
                        srvErrCh <- err
                }
        }()

        bgCtx, bgCancel := context.WithCancel(context.Background())

        c.wp.Start(bgCtx)
        go c.scan.Run(bgCtx)

        quit := make(chan os.Signal, 1)
        signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)

        select {
        case <-quit:
                log.Println("shutting down gracefully")
        case err := <-srvErrCh:
                if !errors.Is(err, http.ErrServerClosed) {
                        log.Printf("server failed to start or crashed: %v", err)
                        bgCancel()
                }
        }

        shutdownCtx, cancel := context.WithTimeout(context.Background(), c.cfg.ShutdownTimeout)
        defer cancel()

        if err := c.srv.Shutdown(shutdownCtx); err != nil {
                log.Printf("Server forced to shutdown: %v", err)
        }
        bgCancel()
        c.wp.Stop()
        if c.tracer != nil {
                defer c.tracer.Shutdown(shutdownCtx)
        }
        c.db.Close()
        log.Println("Server exited gracefully")
}

var gatewayTracer = otel.Tracer("push_gateway")

type Gateway struct {
        client  *http.Client
        baseURL string
}

func NewGateway(baseURl string, timeout time.Duration) *Gateway {
        return &Gateway{
                baseURL: baseURl,
                client: &http.Client{
                        Timeout: timeout,
                },
        }
}

func (g *Gateway) ReuestPosition(ctx context.Context, tripId, driverId uuid.UUID) error {
        ctx, span := gatewayTracer.Start(ctx, "push_gateway.request_position",
                trace.WithAttributes(
                        attribute.String("trip.id", tripId.String()),
                        attribute.String("driver.id", driverId.String()),
                ),
        )
        defer span.End()

        reqBody := pushapi.PushRequest{
                RecipientId: driverId,
                RequestId:   uuid.NewString(),
                Kind:        pushapi.REQUESTPOSITION,
                Data: &map[string]string{
                        "trip_id": tripId.String(),
                },
        }

        bodyBytes, err := json.Marshal(reqBody)
        if err != nil {
                return fmt.Errorf("marshal push request: %w", err)
        }

        req, err := http.NewRequestWithContext(ctx, http.MethodPost, g.baseURL+"/api/v1/push", bytes.NewReader(bodyBytes))
        if err != nil {
                span.SetAttributes(attribute.String("error.type", fmt.Sprintf("%T", err)))
                span.SetStatus(codes.Error, err.Error())
                return fmt.Errorf("create http request: %w", err)
        }
        req.Header.Set("Content-Type", "application/json")

        resp, err := g.client.Do(req)
        if err != nil {
                return fmt.Errorf("transport error: %w", err)
        }
        defer resp.Body.Close()

        switch resp.StatusCode {
        case http.StatusAccepted:
                _, _ = io.Copy(io.Discard, resp.Body)
                return nil
        case http.StatusBadRequest:
                return fmt.Errorf("push service bad request (400)")
        case http.StatusTooManyRequests:
                return fmt.Errorf("push service rate limited (429)")
        case http.StatusServiceUnavailable:
                return fmt.Errorf("push service         unavailable (503)")
        default:
                body, _ := io.ReadAll(resp.Body)
                return fmt.Errorf("push service unexpected status %d: %s", resp.StatusCode, string(body))
        }
}

func Problem(err error, r *http.Request) (int, api.Problem) {
        if err == nil {
                return http.StatusOK, api.Problem{}
        }
        host := getRoutePattern(r)
        typeConflict := "about:blank"
        switch {
        case isValidationError(err):
                detail := "invalid request body or parameters"
                return http.StatusBadRequest, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusBadRequest),
                        Status:   http.StatusBadRequest,
                        Detail:   &detail,
                        Code:     "invalid_request",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrDriverBusy):
                detail := "driver already has active trip"
                return http.StatusConflict, api.CreateTripConflict{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusConflict),
                        Status:   http.StatusConflict,
                        Detail:   &detail,
                        Code:     "driver_busy",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrTripCompleted):
                detail := "trip is already over"
                return http.StatusConflict, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusConflict),
                        Status:   http.StatusConflict,
                        Detail:   &detail,
                        Code:     "trip_completed",
                        Instance: &host,
                }
        case errors.Is(err, trip.ErrNotFound):
                detail := "trip with an id does not exist"
                return http.StatusNotFound, api.BadRequest{
                        Type:     typeConflict,
                        Title:    http.StatusText(http.StatusNotFound),
                        Status:   http.StatusNotFound,
                        Detail:   &detail,
                        Code:     "trip_not_found",
                        Instance: &host,
                }
        }

        detail := "internal server error"
        return http.StatusInternalServerError, api.InternalError{
                Type:     typeConflict,
                Title:    http.StatusText(http.StatusInternalServerError),
                Status:   http.StatusInternalServerError,
                Detail:   &detail,
                Code:     "internal_error",
                Instance: &host,
        }
}

func isValidationError(err error) bool {
        if err == nil {
                return false
        }
        var validateErr *decode.ValidationErr
        if errors.As(err, &validateErr) {
                return true
        }
        var reqErr *openapi3filter.RequestError
        if errors.As(err, &reqErr) {
                return true
        }
        var paramErr *api.InvalidParamFormatError
        return errors.As(err, &paramErr)
}

func getRoutePattern(r *http.Request) string {
        if pattern, ok := r.Context().Value(RoutePatternKey).(string); ok && pattern != "" {
                return pattern
        }

        if routeCtx := chi.RouteContext(r.Context()); routeCtx != nil {
                return routeCtx.RoutePattern()
        }

        return r.URL.Path
}

func Logging(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        lrw := &statusResponseWriter{ResponseWriter: w, status: http.StatusOK}

                        defer func() {
                                var recoveredPanic any
                                if p := recover(); p != nil {
                                        recoveredPanic = p
                                        lrw.status = http.StatusInternalServerError
                                }
                                duration := time.Since(start)
                                level := slog.LevelInfo
                                if lrw.status >= 500 {
                                        level = slog.LevelError
                                }

                                routePattern := ""
                                if routeCtx := chi.RouteContext(r.Context()); routeCtx != nil {
                                        routePattern = routeCtx.RoutePattern()
                                }

                                logger.LogAttrs(r.Context(), level, "request handled",
                                        slog.String("service", "trip-service"),
                                        slog.String("method", r.Method),
                                        slog.String("route", routePattern),
                                        slog.Int("status", lrw.status),
                                        slog.Int64("duration_ms", duration.Milliseconds()),
                                )
                                if recoveredPanic != nil {
                                        panic(recoveredPanic)
                                }
                        }()
                        next.ServeHTTP(lrw, r)
                })
        }
}

type statusResponseWriter struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (lrw *statusResponseWriter) WriteHeader(code int) {
        if !lrw.wroteHeader {
                lrw.status = code
                lrw.wroteHeader = true
                lrw.ResponseWriter.WriteHeader(code)
        }
}

func (lrw *statusResponseWriter) Write(b []byte) (int, error) {
        if !lrw.wroteHeader {
                lrw.WriteHeader(http.StatusOK)
        }
        return lrw.ResponseWriter.Write(b)
}

func Metrics(meter metric.Meter) func(http.Handler) http.Handler {
        activeReq, _ := meter.Int64UpDownCounter(
                "http.server.active_requests",
                metric.WithUnit("{request}"),
                metric.WithDescription("how many requests are being processed right now"),
        )

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        routePattern := chi.RouteContext(r.Context()).RoutePattern()
                        activeReq.Add(r.Context(), 1, metric.WithAttributes(
                                attribute.String("http.request.method", r.Method),
                                attribute.String("http.route", routePattern),
                        ))

                        defer func() {
                                if p := recover(); p != nil {
                                        panic(p)
                                }
                                activeReq.Add(r.Context(), -1, metric.WithAttributes(
                                        attribute.String("http.request.method", r.Method),
                                        attribute.String("http.route", routePattern),
                                ))
                        }()
                        lrw := &statusResponseWriter{ResponseWriter: w, status: http.StatusOK}
                        next.ServeHTTP(lrw, r)
                })
        }
}

func OpenAPIRequestValidator(swagger *openapi3.T) (func(http.Handler) http.Handler, error) {
        router, err := gorillamux.NewRouter(swagger)
        if err != nil {
                return nil, fmt.Errorf("failed create OpenAPI router: %w", err)
        }

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        route, pathParams, err := router.FindRoute(r)
                        if err != nil {
                                next.ServeHTTP(w, r)
                                return
                        }

                        requestValidationInput := &openapi3filter.RequestValidationInput{
                                Request:    r,
                                PathParams: pathParams,
                                Route:      route,
                        }

                        if err := openapi3filter.ValidateRequest(r.Context(), requestValidationInput); err != nil {
                                ctx := context.WithValue(r.Context(), domainerror.RoutePatternKey, route.Path)
                                r = r.WithContext(ctx)
                                response.HandleAPIError(w, r, err)
                                return
                        }
                        next.ServeHTTP(w, r)
                })
        }, nil
}

        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        defer func() {
                                if err := recover(); err != nil {
                                        logger.ErrorContext(r.Context(), "panic recovered",
                                                "error", err,
                                                "stack", string(debug.Stack()),
                                        )

                                        status, problem := domainerror.Problem(errors.New("internal server error"), r)
                                        response.Problem(w, status, problem)
                                }
                        }()
                        next.ServeHTTP(w, r)
                })
        }
}

func NewDB(ctx context.Context, cfg *config.Config) (*DB, error) {
        pgxCfg, err := pgxpool.ParseConfig(cfg.DatabaseUrl)
        if err != nil {
                return nil, fmt.Errorf("parse pgx config: %w", err)
        }
        pgxCfg.MaxConns = int32(cfg.DatabaseMaxConns) //nolint:gosec
        pgxCfg.MinConns = int32(cfg.DatabaseMinConns) //nolint:gosec
        pgxCfg.MaxConnLifetime = cfg.DatabaseMaxConnsLifetime
        pgxCfg.ConnConfig.ConnectTimeout = cfg.DatabaseConnectTimeout

        if err != nil {
                return nil, fmt.Errorf("create pool: %w", err)
        }

        pingCtx, pingCancel := context.WithTimeout(ctx, cfg.DatabaseConnectTimeout)
        defer pingCancel()
        if err = pool.Ping(pingCtx); err != nil {
                pool.Close()
                return nil, fmt.Errorf("connection from pool: %w", err)
        }

        return &DB{Pool: pool}, nil
}

        if db.Pool != nil {
                db.Pool.Close()
        }
}

package txManager

import (
        "context"
        "fmt"
        "time"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"
)

type contextKey string

const TxKey contextKey = "transaction"

type TransactionManager struct {
        pool      *pgxpool.Pool
        opTimeout time.Duration
}

func NewTransactionManager(pool *pgxpool.Pool, opTimeout time.Duration) *TransactionManager {
        return &TransactionManager{
                pool:      pool,
                opTimeout: opTimeout,
        }
}

func (tm *TransactionManager) Do(ctx context.Context, fn func(ctx context.Context) error) (err error) {
        if _, ok := ctx.Value(TxKey).(pgx.Tx); ok {
                return fn(ctx)
        }

        beginCtx, beginCancel := context.WithTimeout(ctx, tm.opTimeout)
        tx, err := tm.pool.Begin(beginCtx)
        beginCancel()

        if err != nil {
                return fmt.Errorf("starts a transaction: %w", err)
        }

        defer func() {
                if p := recover(); p != nil {
                        rbCtx, rbCancel := context.WithTimeout(context.WithoutCancel(ctx), tm.opTimeout)
                        _ = tx.Rollback(rbCtx)
                        rbCancel()
                        panic(p)
                } else if err != nil {
                        rbCtx, rbCancel := context.WithTimeout(context.WithoutCancel(ctx), tm.opTimeout)
                        if rbErr := tx.Rollback(rbCtx); rbErr != nil {
                                err = fmt.Errorf("operation failed: %w, rollback failed: %v", err, rbErr)
                        }
                        rbCancel()
                } else {
                        cmCtx, cmCancel := context.WithTimeout(ctx, tm.opTimeout)
                        err = tx.Commit(cmCtx)
                        cmCancel()
                }
        }()

        ctx = context.WithValue(ctx, TxKey, tx)
        return fn(ctx)
}

func NewLogger(level slog.Level) *slog.Logger {
        opts := &slog.HandlerOptions{
                Level: level,
                ReplaceAttr: func(groups []string, a slog.Attr) slog.Attr {
                        if a.Key == slog.TimeKey {
                                if t, ok := a.Value.Any().(time.Time); ok {
                                        a.Value = slog.StringValue(t.UTC().Format("2006-01-02T15:04:05.000Z"))
                                }
                        }
                        return a
                },
        }

        handler := slog.NewJSONHandler(os.Stdout, opts)
        return slog.New(&traceHandler{inner: handler})
}

        return h.inner.Handle(ctx, r)

type OTelProviders struct {
        Tracer *trace.TracerProvider
        Meter  *metric.MeterProvider
}

func NewOTel(ctx context.Context, cfg *config.Config) (*OTelProviders, error) {
        p := &OTelProviders{}

        res, err := resource.New(ctx,
                resource.WithFromEnv(),
                resource.WithAttributes(
                        semconv.ServiceNameKey.String(cfg.OtelServiceName),
                ),
        )
        if err != nil {
                return nil, err
        }

        endpointHost := cfg.OtelExporterEndpoint
        if u, err := url.Parse(cfg.OtelExporterEndpoint); err == nil && u.Host != "" {
                endpointHost = u.Host
        }

        traceExp, err := otlptracehttp.New(ctx,
                otlptracehttp.WithEndpoint(endpointHost),
                otlptracehttp.WithInsecure(),
        )
        if err != nil {
                slog.Warn("trace exporter init failed, traces disabled", "error", err)
        } else {
                p.Tracer = trace.NewTracerProvider(
                        trace.WithResource(res),
                        trace.WithBatcher(traceExp),
                        trace.WithSampler(trace.ParentBased(trace.AlwaysSample())),
                )
                otel.SetTracerProvider(p.Tracer)
        }

        metricExp, err := otlpmetrichttp.New(ctx,
                otlpmetrichttp.WithEndpoint(endpointHost),
                otlpmetrichttp.WithInsecure(),
        )
        if err != nil {
                slog.Warn("metric exporter init failed", "error", err)
        } else {
                p.Meter = metric.NewMeterProvider(
                        metric.WithResource(res),
                        metric.WithReader(metric.NewPeriodicReader(metricExp)),
                )
                otel.SetMeterProvider(p.Meter)
        }

        otel.SetTextMapPropagator(propagation.NewCompositeTextMapPropagator(
                propagation.TraceContext{},
                propagation.Baggage{},
        ))

        return p, nil
}

func (tp *OTelProviders) Shutdown(ctx context.Context) {
        if tp.Tracer != nil {
                _ = tp.Tracer.Shutdown(ctx)
        }
        if tp.Meter != nil {
                _ = tp.Meter.Shutdown(ctx)
        }
}

type dbExecutor interface {
        Exec(ctx context.Context, sql string, arguments ...any) (commandTag pgconn.CommandTag, err error)
        QueryRow(ctx context.Context, sql string, args ...any) pgx.Row
        Query(ctx context.Context, sql string, args ...any) (pgx.Rows, error)
}

func getDbExecutor(ctx context.Context, pool *pgxpool.Pool) dbExecutor {
        if tx, ok := ctx.Value(txManager.TxKey).(pgx.Tx); ok {
                return tx
        }
        return pool
}

func (r *Repository) Ping(ctx context.Context) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        return r.pool.Ping(queryCtx)
}

func (r *Repository) Create(ctx context.Context, t *trip.Trip) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)
        span := trace.SpanFromContext(queryCtx)

        tripBuilder := sq.Insert("trips").
                Columns(
                        "id", "user_id", "driver_id",
                        "start_latitude", "start_longitude", "end_latitude", "end_longitude",
                        "price", "status",
                        "started_at", "created_at", "updated_at",
                ).
                Values(
                        t.Id, t.UserId, t.DriverId,
                        t.StartPoint.Lat, t.StartPoint.Lon, t.EndPoint.Lat, t.EndPoint.Lon,
                        t.Price, t.Status,
                        t.StartedAt, time.Now(), time.Now(),
                ).
                PlaceholderFormat(sq.Dollar)

        query, args, err := tripBuilder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip query: %w", err)
        }

        span.SetAttributes(
                attribute.String("db.operation.name", "INSERT"),
                attribute.String("db.collection.name", "trips"),
        )

        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                span.SetAttributes(attribute.String("error.type", fmt.Sprintf("%T", err)))
                span.SetStatus(codes.Error, err.Error())
                var pgErr *pgconn.PgError
                if errors.As(err, &pgErr) && pgErr.Code == "23505" {
                        return trip.ErrDriverBusy
                }
                return fmt.Errorf("failed exec trips: %w", err)
        }

        historyBulder := sq.Insert("trip_status_history").
                Columns("trip_id", "from_status", "to_status", "reason", "changed_at").
                Values(t.Id, nil, t.Status, "trip created", t.StartedAt).
                PlaceholderFormat(sq.Dollar)

        query, args, err = historyBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip_status_history query: %w", err)
        }

        span.SetAttributes(
                attribute.String("db.operation.name", "INSERT"),
                attribute.String("db.collection.name", "trip_status_history"),
        )

        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                span.SetAttributes(attribute.String("error.type", fmt.Sprintf("%T", err)))
                span.SetStatus(codes.Error, err.Error())
                return fmt.Errorf("failed exec trip_status_history: %w", err)
        }

        return nil
}

func (r *Repository) Save(ctx context.Context, t *trip.Trip) error {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)
        span := trace.SpanFromContext(queryCtx)

        oldBulder := sq.Select("status").
                From("trips").
                Where(sq.Eq{"id": t.Id}).
                PlaceholderFormat(sq.Dollar)
        query, args, err := oldBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde old_trip query: %w", err)
        }

        span.SetAttributes(
                attribute.String("db.operation.name", "SELECT"),
                attribute.String("db.collection.name", "trips"),
        )

        var oldStatus string
        err = dbExc.QueryRow(queryCtx, query, args...).Scan(&oldStatus)
        if err != nil {
                span.SetAttributes(attribute.String("error.type", fmt.Sprintf("%T", err)))
                span.SetStatus(codes.Error, err.Error())
                if errors.Is(err, pgx.ErrNoRows) {
                        return trip.ErrNotFound
                }
                return fmt.Errorf("failed QueryRow old status: %w", err)
        }

        tripBulder := sq.Update("trips").
                Set("status", t.Status).
                Set("finished_at", t.FinishedAt).
                Set("updated_at", t.FinishedAt).
                Where(sq.Eq{"id": t.Id}).
                PlaceholderFormat(sq.Dollar)
        query, args, err = tripBulder.ToSql()
        if err != nil {
                return fmt.Errorf("builde trip query: %w", err)
        }

        span.SetAttributes(
                attribute.String("db.operation.name", "UPDATE"),
                attribute.String("db.collection.name", "trips"),
        )

func (r *Repository) SavePosition(ctx context.Context, pos trip.TripPosition) (int64, error) {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)
        //span := trace.SpanFromContext(queryCtx)
        tripBuilder := sq.Select("status").
                From("trips").
                Where(sq.Eq{"id": pos.TripId}).
                PlaceholderFormat(sq.Dollar)

        query, args, err := tripBuilder.ToSql()
        if err != nil {
                return 0, fmt.Errorf("builde trip query: %w", err)
        }
        var status string
        err = dbExc.QueryRow(queryCtx, query, args...).Scan(&status)
        if err != nil {
                if err == pgx.ErrNoRows {
                        return 0, trip.ErrNotFound
                }
                return 0, fmt.Errorf("failed QueryRow trip: %w", err)
        }
        if status != string(trip.Active) {
                return 0, trip.ErrTripCompleted
        }

        positionsBuilder := sq.Insert("trip_positions").
                Columns("trip_id", "latitude", "longitude", "recorded_at").
                Values(pos.TripId, pos.Coordinates.Lat, pos.Coordinates.Lon, pos.RecordedAt).
                Suffix("RETURNING id").
                PlaceholderFormat(sq.Dollar)
        query, args, err = positionsBuilder.ToSql()
        if err != nil {
                return 0, fmt.Errorf("builde position query: %w", err)
        }
        var posId int64
        err = dbExc.QueryRow(queryCtx, query, args...).Scan(&posId)
        if err != nil {
                return 0, fmt.Errorf("failed insert position: %w", err)
        }

        lastPosBuilder := sq.Update("trips").
                Set("last_position_at", sq.Expr(
                        "GREATEST(COALESCE(last_position_at, '1970-01-01 00:00:00+00'::timestamptz), ?)",
                        pos.RecordedAt,
                )).
                Where(sq.Eq{"id": pos.TripId}).
                PlaceholderFormat(sq.Dollar)
        query, args, err = lastPosBuilder.ToSql()
        if err != nil {
                return 0, fmt.Errorf("builde last position query: %w", err)
        }
        _, err = dbExc.Exec(queryCtx, query, args...)
        if err != nil {
                return 0, fmt.Errorf("failed update last position: %w", err)
        }
        return posId, nil
}

func (r *Repository) GetPositions(ctx context.Context, tripId uuid.UUID) ([]trip.TripPosition, error) {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)
        //span := trace.SpanFromContext(queryCtx)
        positionsBuilder := sq.Select(
                "id, trip_id, latitude, longitude, recorded_at").

        query, args, err := positionsBuilder.ToSql()
        if err != nil {
                return nil, fmt.Errorf("builde positions query: %w", err)
        }
        rows, err := dbExc.Query(queryCtx, query, args...)
        if err != nil {
                return nil, fmt.Errorf("failed select positions: %w", err)
        }
        defer rows.Close()
        var positions []trip.TripPosition
        for rows.Next() {

func (r *Repository) GetStaleTrips(ctx context.Context, staleAfter time.Duration) ([]trip.StaleTrip, error) {
        queryCtx, cancel := context.WithTimeout(ctx, r.queryTimeout)
        defer cancel()

        dbExc := getDbExecutor(queryCtx, r.pool)
        staleBuilder := sq.Select("id", "driver_id").
                From("trips").
                Where(sq.Eq{"status": trip.Active}).
                Where(sq.Expr(
                        "COALESCE(last_position_at, started_at) < ?",
                        time.Now().UTC().Add(-staleAfter),
                )).
                OrderBy("COALESCE(last_position_at, started_at)").
                PlaceholderFormat(sq.Dollar)

        query, args, err := staleBuilder.ToSql()
        if err != nil {
                return nil, fmt.Errorf("builde stale trips query: %w", err)
        }
        rows, err := dbExc.Query(queryCtx, query, args...)
        if err != nil {
                return nil, fmt.Errorf("failed execute get stale trips query: %w", err)
        }
        defer rows.Close()
        var staleTrips []trip.StaleTrip
        for rows.Next() {
                var staleTrip trip.StaleTrip
                err = rows.Scan(&staleTrip.Id, &staleTrip.DriverId)
                if err != nil {
                        return nil, fmt.Errorf("scan stale trip error: %w", err)
                }
                staleTrips = append(staleTrips, staleTrip)
        }
        if err = rows.Err(); err != nil {
                return nil, fmt.Errorf("iterate stale trips rows: %w", err)
        }
        return staleTrips, nil
}

package currentlyinqueue

import (
        "sync"

        "github.com/google/uuid"
)

type InQueue struct {
        mu    sync.RWMutex
        items map[uuid.UUID]struct{}
}

func NewInQueue() *InQueue {
        return &InQueue{
                items: make(map[uuid.UUID]struct{}),
        }
}

func (q *InQueue) TryAdd(tripId uuid.UUID) bool {
        q.mu.Lock()
        defer q.mu.Unlock()

        if _, ok := q.items[tripId]; ok {
                return false
        }
        q.items[tripId] = struct{}{}
        return true
}

func (q *InQueue) Remove(tripId uuid.UUID) {
        q.mu.Lock()
        defer q.mu.Unlock()
        delete(q.items, tripId)
}

func (q *InQueue) Contains(tripId uuid.UUID) bool {
        q.mu.RLock()
        defer q.mu.RUnlock()
        _, ok := q.items[tripId]
        return ok
}

func (q *InQueue) Len() int {
        q.mu.RLock()
        defer q.mu.RUnlock()
        return len(q.items)
}

var scannerTracer = otel.Tracer("scanner")

type Scanner struct {
        checkInterval time.Duration
        staleAfter    time.Duration
        repo          StaleTripsGetter
        workerPool    *workerpool.WorkerPool
        inQueue       *currentlyinqueue.InQueue
        ticker        Ticker
        logger        *slog.Logger
        metrics       *metrics.ScannerMetrics
}

func NewScanner(
        checkInterval time.Duration,
        staleAfter time.Duration,
        repo StaleTripsGetter,
        workerPool *workerpool.WorkerPool,
        inQueue *currentlyinqueue.InQueue,
        ticker Ticker,
        logger *slog.Logger,
        metrics *metrics.ScannerMetrics,

) *Scanner {
        return &Scanner{
                checkInterval: checkInterval,
                staleAfter:    staleAfter,
                repo:          repo,
                workerPool:    workerPool,
                inQueue:       inQueue,
                ticker:        ticker,
                logger:        logger,
                metrics:       metrics,
        }
}

func (s *Scanner) Run(ctx context.Context) {
        s.logger.InfoContext(ctx, "scanner started",
                slog.Duration("check_interval", s.checkInterval),
                slog.Duration("stale_after", s.staleAfter),
        )
        defer func() {
                s.ticker.Stop()
                s.logger.InfoContext(ctx, "scanner stopped")
        }()

        for {
                select {
                case <-ctx.Done():
                        return
                case <-s.ticker.Chan():
                        s.tick(ctx)
                }
        }
}

func (s *Scanner) tick(ctx context.Context) {
        staleTrips, err := s.repo.GetStaleTrips(ctx, s.staleAfter)
        if err != nil {
                s.logger.WarnContext(ctx, "failed to get stale trips",
                        slog.Any("error", err),
                )
                return
        }

        s.logger.DebugContext(ctx, "stale trips fetched",
                slog.Int("count", len(staleTrips)),
        )

        for _, st := range staleTrips {
                s.enqueue(ctx, st)
        }
}

func (s *Scanner) enqueue(ctx context.Context, st trip.StaleTrip) {
        ctx, span := scannerTracer.Start(ctx, "scanner.enqueue")
        span.SetAttributes(attribute.String("trip.id", st.Id.String()))
        defer span.End()

        if !s.inQueue.TryAdd(st.Id) {
                s.metrics.RecordCheck(ctx, "duplicate")
                return
        }

        st.ScannerSpan = span.SpanContext()
        if !s.workerPool.Submit(st) {
                s.inQueue.Remove(st.Id)
                s.metrics.RecordCheck(ctx, "dropped")
                s.logger.WarnContext(ctx, "queue full, task dropped",
                        slog.String("id", st.Id.String()),
                )
                return
        }
        s.metrics.RecordCheck(ctx, "queued")
}

package workerpool

import (
        "context"
        "log/slog"
        "sync"
        "sync/atomic"
        "time"

        "github.com/adeladnagulov/TripGo/internal/domain/trip"
        currentlyinqueue "github.com/adeladnagulov/TripGo/internal/worker/positions/currently_in_queue"
        "github.com/adeladnagulov/TripGo/internal/worker/positions/metrics"
        "github.com/google/uuid"
        "go.opentelemetry.io/otel"
        "go.opentelemetry.io/otel/attribute"
        "go.opentelemetry.io/otel/trace"
)

var workerTracer = otel.Tracer("worker_pool")

type Gateway interface {
        ReuestPosition(ctx context.Context, tripId, driverId uuid.UUID) error
}

type WorkerPool struct {
        queue       chan trip.StaleTrip
        inQueue     *currentlyinqueue.InQueue
        pushGateway Gateway
        pushTimeout time.Duration
        workers     int
        logger      *slog.Logger

        metrics       *metrics.PoolMetrics
        activeWorkers atomic.Int64
        wg            sync.WaitGroup
}

func NewWorkerPool(
        queueSize int,
        inQueue *currentlyinqueue.InQueue,
        pushGateway Gateway,
        pushTimeout time.Duration,
        workers int,
        logger *slog.Logger,
        poolMetrics *metrics.PoolMetrics,
) *WorkerPool {
        return &WorkerPool{
                queue:       make(chan trip.StaleTrip, queueSize),
                inQueue:     inQueue,
                pushGateway: pushGateway,
                pushTimeout: pushTimeout,
                workers:     workers,
                logger:      logger,
                metrics:     poolMetrics,
        }
}

func (wp *WorkerPool) Submit(st trip.StaleTrip) bool {
        select {
        case wp.queue <- st:
                return true
        default:
                return false
        }
}

func (wp *WorkerPool) Start(ctx context.Context) {
        for i := 0; i < wp.workers; i++ {
                wp.wg.Add(1)
                go wp.worker(ctx, i)
        }
}

func (wp *WorkerPool) Stop() {
        close(wp.queue)
        wp.wg.Wait()
}

func (wp *WorkerPool) worker(ctx context.Context, id int) {
        defer wp.wg.Done()
        wp.logger.DebugContext(ctx, "worker started", slog.Int("worker_id", id))
        defer wp.logger.DebugContext(ctx, "worker stopped", slog.Int("worker_id", id))

        for st := range wp.queue {
                wp.processTrip(ctx, st, id)
        }
}

func (wp *WorkerPool) processTrip(ctx context.Context, st trip.StaleTrip, workerId int) {
        if !wp.inQueue.Contains(st.Id) {
                wp.metrics.RecordRequest(ctx, "cancelled")
                wp.logger.DebugContext(ctx, "task cancelled, skipping",
                        slog.String("id", st.Id.String()),
                        slog.Int("worker_id", workerId),
                )
                return
        }

        wp.activeWorkers.Add(1)
        defer wp.activeWorkers.Add(-1)

        spanCtx, span := workerTracer.Start(ctx, "worker.process_trip",
                trace.WithLinks(trace.Link{
                        SpanContext: st.ScannerSpan,
                }),
        )
        span.SetAttributes(attribute.String("trip.id", st.Id.String()))
        defer span.End()

        pushCtx, cancel := context.WithTimeout(spanCtx, wp.pushTimeout)
        defer cancel()

        start := time.Now()
        err := wp.pushGateway.ReuestPosition(pushCtx, st.Id, st.DriverId)
        duration := time.Since(start)

        wp.inQueue.Remove(st.Id)

        result := "success"
        if err != nil {
                result = "error"
                wp.logger.WarnContext(ctx, "push request failed",
                        slog.String("error", err.Error()),
                        slog.String("id", st.Id.String()),
                        slog.Int("worker_id", workerId),
                )
        } else {
                wp.logger.InfoContext(ctx, "push request sent",
                        slog.String("id", st.Id.String()),
                        slog.Int("worker_id", workerId),
                )
        }

        wp.metrics.RecordRequest(ctx, result)
        wp.metrics.RecordDuration(ctx, duration, result)
}

func (wp *WorkerPool) GetActiveWorkers() int64 {
        return wp.activeWorkers.Load()
}

func (wp *WorkerPool) GetQueueSize() int {
        return len(wp.queue)
}
